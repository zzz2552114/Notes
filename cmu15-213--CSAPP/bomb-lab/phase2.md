***反汇编后的代码如下，注意对照注释查看***
```x86asm
(gdb) disassemble phase_2
Dump of assembler code for function phase_2:
   // %rbp和%rbx是被调用者有义务保护的寄存器
   0x0000000000400efc <+0>:     push   %rbp
   0x0000000000400efd <+1>:     push   %rbx
   // 0x28 = 40，这里扩展了40字节的栈空间
   0x0000000000400efe <+2>:     sub    $0x28,%rsp
   // 把%rsp这个指针竟然赋值给了%rsi，说明这段空间要被利用了
   0x0000000000400f02 <+6>:     mov    %rsp,%rsi
   // 调用函数，%rdi没变，说明还是我们的input，下面我们去看一下这个函数
   0x0000000000400f05 <+9>:     call   0x40145c <read_six_numbers>
   // 由于%rsp也属于被调用者有义务保存的，所以这里%rsp不动。而且由我们对上一个函数的分析，我们知道这里解址后应该是输入的第一个整数。在判断它是不是1
   0x0000000000400f0a <+14>:    cmpl   $0x1,(%rsp)
   // 是1跳转，不是1就爆炸，直接去看+52
   0x0000000000400f0e <+18>:    je     0x400f30 <phase_2+52>
   0x0000000000400f10 <+20>:    call   0x40143a <explode_bomb>
   0x0000000000400f15 <+25>:    jmp    0x400f30 <phase_2+52>
   // 时刻注意lea和mov中“(寄存器)”这个运算的不同！！！这里是取上一个int值。
   0x0000000000400f17 <+27>:    mov    -0x4(%rbx),%eax
   // 倍增
   0x0000000000400f1a <+30>:    add    %eax,%eax
   // 比较这个值和2*上个值是不是相等，不相等就爆炸
   0x0000000000400f1c <+32>:    cmp    %eax,(%rbx)
   // 跳转+41
   0x0000000000400f1e <+34>:    je     0x400f25 <phase_2+41>
   0x0000000000400f20 <+36>:    call   0x40143a <explode_bomb>
   // +4成为下一个元素的指针
   0x0000000000400f25 <+41>:    add    $0x4,%rbx
   // 还记得%rbp吗？判断是不是越界！
   0x0000000000400f29 <+45>:    cmp    %rbp,%rbx
   // 不越界就继续回到+27，形成do while循环
   0x0000000000400f2c <+48>:    jne    0x400f17 <phase_2+27>
   // 越界就去+64，我们去看看
   0x0000000000400f2e <+50>:    jmp    0x400f3c <phase_2+64>
   // 刚刚跳转到这里，是让%rdx作为下一个int的指针
   0x0000000000400f30 <+52>:    lea    0x4(%rsp),%rbx
   // 这里可以算一下，0x18=24，从0开始计数那么这是越界的第一个地址
   0x0000000000400f35 <+57>:    lea    0x18(%rsp),%rbp
   // 继续跳转到+27
   0x0000000000400f3a <+62>:    jmp    0x400f17 <phase_2+27>
   // 这里是+64，在收拾后事了，至此没有引爆炸弹，成功了
   0x0000000000400f3c <+64>:    add    $0x28,%rsp
   0x0000000000400f40 <+68>:    pop    %rbx
   0x0000000000400f41 <+69>:    pop    %rbp
   0x0000000000400f42 <+70>:    ret
```

```x86asm
(gdb) disassemble read_six_numbers
Dump of assembler code for function read_six_numbers:
   // 扩展了这个函数的栈空间，要做什么暂时还不知道
   0x000000000040145c <+0>:     sub    $0x18,%rsp
   // %rdx是可变的寄存器，要干什么不知道，但是要记住%rsi是指向上个函数的可用空间的指针
   0x0000000000401460 <+4>:     mov    %rsi,%rdx
   // 要记住lea里的(%rsi)只是拿地址，而不是访问地址！！！这里拿的是：上个函数可用空间的第4个字节地址(从0计数)。这个时候就可以推测是int了，四字节的东西不多
   0x0000000000401463 <+7>:     lea    0x4(%rsi),%rcx
   // 这个地方很傻福，直接跳到+20字节了，暂时不看，后面再看
   0x0000000000401467 <+11>:    lea    0x14(%rsi),%rax
   0x000000000040146b <+15>:    mov    %rax,0x8(%rsp)
   // +16，也后面再看
   0x0000000000401470 <+20>:    lea    0x10(%rsi),%rax
   0x0000000000401474 <+24>:    mov    %rax,(%rsp)

   0x0000000000401478 <+28>:    lea    0xc(%rsi),%r9
   // +8，放到参数寄存器里
   0x000000000040147c <+32>:    lea    0x8(%rsi),%r8

   0x000000000040148f <+51>:    cmp    $0x5,%eax

   0x0000000000401494 <+56>:    call   0x40143a <explode_bomb>
   0x0000000000401499 <+61>:    add    $0x18,%rsp
   // 返回值并不重要，可以回到上面的函数了
   0x000000000040149d <+65>:    ret
```
